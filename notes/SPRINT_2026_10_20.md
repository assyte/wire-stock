### EAN-13 Checksum

```typescript
function generateEAN13(): string {
  const base = Array.from({ length: 12 }, () => Math.floor(Math.random() * 10));
  const sum = base.reduce((acc, digit, i) => acc + digit * (i % 2 === 0 ? 1 : 3), 0);
  const checksum = (10 - (sum % 10)) % 10;
  return [...base, checksum].join('');
}

function isValidEAN13(barcode: string): boolean {
  if (!/^\d{13}$/.test(barcode)) return false;
  const digits = barcode.split('').map(Number);
  const sum = digits.slice(0, 12).reduce((acc, digit, i) => acc + digit * (i % 2 === 0 ? 1 : 3), 0);
  const expected = (10 - (sum % 10)) % 10;
  return expected === digits[12];
}
```
